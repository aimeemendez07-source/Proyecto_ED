cmake_minimum_required(VERSION 3.21)
project(Puzzle8 LANGUAGES CXX)

# Asegura la compatibilidad con librerías externas
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)

# Configura el modo Debug por defecto si no se especifica
if(NOT CMAKE_BUILD_TYPE)
	set(CMAKE_BUILD_TYPE Debug CACHE STRING "Tipo de compilación" FORCE)
endif()

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

#descarga raylib automáticamente desde github
include(FetchContent)

FetchContent_Declare(
    raylib
    GIT_REPOSITORY https://github.com/raysan5/raylib.git
    GIT_TAG 5.5
    GIT_SHALLOW TRUE
)

set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(raylib)

add_executable(puzzle8
    src/main.cpp
)

target_include_directories(puzzle8 PRIVATE include)
target_link_libraries(puzzle8 PRIVATE raylib)
